    }
];

let cart = [];

function displayFoods(items) {
    const menu = document.getElementById("foodMenu");

    menu.innerHTML = items.map((food) => `
        <div class="food-card">
            <img src="${food.image}" alt="${food.name}"
                 onerror="this.src='https://placehold.co/500x300?text=Food'">
            <div class="food-info">
                <h3>${food.name}</h3>
                <p>${food.description}</p>
                <div class="price">Rs. ${food.price}</div>
                <button class="btn" onclick="addToCart('${food.name}')">
                    Add to Cart +
                </button>
            </div>
        </div>
    `).join("");
}

function addToCart(name) {
    const food = foods.find(item => item.name === name);
    const existing = cart.find(item => item.name === name);

    if (existing) {
        existing.quantity++;
    } else {
        cart.push({...food, quantity: 1});
    }

    updateCart();
}

function updateCart() {
    const cartItems = document.getElementById("cartItems");

    if (cart.length === 0) {
        cartItems.innerHTML = "Your cart is empty.";
    } else {
        cartItems.innerHTML = cart.map((item, index) => `
            <div class="cart-item">
                <span>
                    ${item.name} x ${item.quantity}
                </span>
                <span>
                    Rs. ${item.price * item.quantity}
                    <button onclick="removeItem(${index})">❌</button>
                </span>
            </div>
        `).join("");
    }

    let total = cart.reduce(
        (sum, item) => sum + item.price * item.quantity, 0
    );

    document.getElementById("total").innerText =
        "Total: Rs. " + total;
}

function removeItem(index) {
    cart.splice(index, 1);
    updateCart();
}

function checkout() {
    if (cart.length === 0) {
        alert("Please add food to your cart first!");
        return;
    }

    let message = "Hello! I want to place an order:\n\n";

    cart.forEach(item => {
        message += `${item.name} x ${item.quantity} = Rs. ${item.price * item.quantity}\n`;
    });

    let total = cart.reduce(
        (sum, item) => sum + item.price * item.quantity, 0
    );

    message += `\nTotal: Rs. ${total}`;

    const phone = "923001234567";

    window.open(
        "https://wa.me/" + phone + "?text=" +
        encodeURIComponent(message),
        "_blank"
    );
}

document.getElementById("search").addEventListener("input", function() {
    const query = this.value.toLowerCase();

    const filtered = foods.filter(food =>
        food.name.toLowerCase().includes(query)
    );

    displayFoods(filtered);
});

displayFoods(foods);
